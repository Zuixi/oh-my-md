# M1 引擎 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付可运行的 oh-my-md v0.1：Tauri 2 桌面壳 + CM6 编辑引擎，支持 GFM 全量解析、Live Preview 所见即所得（行内语法折叠）、live/source 模式切换、打开/保存文件。

**Architecture:** pnpm monorepo，两个包：`packages/engine`（纯 TS 编辑引擎，装饰管线是纯函数，可无 DOM 单测）+ `apps/desktop`（Tauri 2 + React 壳）。渲染策略 B：行内语法用 decorations 折叠语法符号；块级 widget（数学/图表/表格渲染）属 M2，本计划中表格只做解析验证。

**Tech Stack:** TypeScript, Tauri 2 (Rust), React, CodeMirror 6, @lezer/markdown (GFM + Footnotes), Vitest, pnpm workspaces。

**范围边界（与 spec 对齐）：**
- 表格在 M1 只做**解析**验证；TableWidget 渲染归 M2。
- 图片、KaTeX、Mermaid、代码高亮归 M2；标签页/文件树/导出/主题归 M3。
- M1 交付物：打开单个 .md（Cmd+O）→ 编辑（Live Preview）→ 保存（Cmd+S），模式切换 Cmd+E。

`packages/engine/src/decorations/build.ts`:
```ts
import { EditorState, Range } from "@codemirror/state"
import { syntaxTree } from "@codemirror/language"
import { Decoration, DecorationSet, EditorView, ViewPlugin, ViewUpdate } from "@codemirror/view"
import { inlineRules } from "./inline"
import { blockRules } from "./blocks"

export interface DecoSpec { from: number; to: number; tag: string; deco: Decoration }

// 光标所在的语法标记不折叠（否则光标进不去、IME 会抖）
export function nearCursor(state: EditorState, from: number, to: number) {
  const { from: sf, to: st } = state.selection.main
  return from < st && to > sf
}

export function collectDecorationSpecs(state: EditorState, from: number, to: number): DecoSpec[] {
  const out: DecoSpec[] = []
  syntaxTree(state).iterate({
    from, to,
    enter(node) {
      inlineRules(node, state, out)
      blockRules(node, state, out)
    }
  })
  return out
}

export function buildLiveDecorations(state: EditorState, from: number, to: number): DecorationSet {
  const ranges: Range<Decoration>[] = collectDecorationSpecs(state, from, to).map(s => s.deco.range(s.from, s.to))
  return Decoration.set(ranges, true)
}

export const livePreviewPlugin = ViewPlugin.fromClass(class {
  decorations: DecorationSet
  constructor(view: EditorView) {
    this.decorations = buildLiveDecorations(view.state, view.viewport.from, view.viewport.to)
  }
  update(u: ViewUpdate) {
    if (u.docChanged || u.viewportChanged || u.selectionSet)
      this.decorations = buildLiveDecorations(u.view.state, u.view.viewport.from, u.view.viewport.to)
  }
}, { decorations: v => v.decorations })
```

`packages/engine/src/decorations/inline.ts`（本任务只实现标题，其余规则在 Task 4 补全，先给空骨架）:
```ts
import type { SyntaxNodeRef } from "@lezer/common"
import { EditorState } from "@codemirror/state"
import { Decoration } from "@codemirror/view"
import { nearCursor, type DecoSpec } from "./build"

const HEADING_CLASS: Record<string, string> = {
  ATXHeading1: "omd-h1", ATXHeading2: "omd-h2", ATXHeading3: "omd-h3",
  ATXHeading4: "omd-h4", ATXHeading5: "omd-h5", ATXHeading6: "omd-h6",
}

export function inlineRules(node: SyntaxNodeRef, state: EditorState, out: DecoSpec[]) {
  const headingClass = HEADING_CLASS[node.name]
  if (headingClass) {
    out.push({ from: node.from, to: node.from, tag: `line:${headingClass}`, deco: Decoration.line({ class: headingClass }) })
    // 折叠 HeaderMark（'#' 及其后空格）
    const cursor = node.node.cursor()
    if (cursor.firstChild()) {
      do {
        if (cursor.name === "HeaderMark") {
          const end = Math.min(cursor.to + 1
, state.doc.lineAt(node.from).to)  // 含后随空格
          if (!nearCursor(state, cursor.from, end))
            out.push({ from: cursor.from, to: end, tag: "replace:HeaderMark", deco: Decoration.replace({}) })
        }
      } while (cursor.nextSibling())
    }
  }
}
```

在 `inline.ts` 的 `inlineRules` 中追加（保留标题逻辑）:
```ts
function childMark(node: SyntaxNodeRef, name: string): { from: number; to: number }[] {
  const marks: { from: number; to: number }[] = []
  const c = node.node.cursor()
  if (c.firstChild()) do { if (c.name === name) marks.push({ from: c.from, to: c.to }) } while (c.nextSibling())
  return marks
}

// 折叠成对语法标记的首尾（**bold** / *em* / ~~del~~）
function foldPair(node: SyntaxNodeRef, state: EditorState, out:
 DecoSpec[], markClass: string) {
  const marks = childMark(node, "CodeMark")
  if (marks.length < 2) return
  for (const m of [marks[0], marks[marks.length - 1]])
    if (!nearCursor(state, m.from, m.to))
      out.push({ from: m.from, to: m.to, tag: "replace:CodeMark", deco: Decoration.replace({}) })
  out.push({ from: node.from, to: node.to, tag: `mark:${markClass}`, deco: Decoration.mark({ class: markClass }) })
}

const EMPHASIS_CLASS: Record<string, string> = {
  StrongEmphasis: "omd-strong", Emphasis: "omd-em", Strikethrough: "omd-del",
}
```

然后在 `inlineRules` 开头（标题判断之后）分发：
```ts
  const emClass = EMPHASIS_CLASS[node.name]
  if (emClass) { foldPair(node, state, out, emClass); return }

  if (node.name === "InlineCode") {
    const marks = childMark(node, "CodeMark")
    for (const m of marks)
      if (!nearCursor(state, m.from, m.to))
        out.push({ from: m.from, to: m.to, tag: "replace:CodeMark", deco: Decoration.replace({}) })
    out.push({ from: node.from, to: node.to, tag: "mark:omd-inline-code", deco: Decoration.mark({ class: "omd-inline-code" }) })
    return
  }

  if (node.name === "Link") {
    // 折叠全部 LinkMark（[ ] ( )）
    const linkMarks = childMark(node, "LinkMark")
    let lFrom = Infinity, lTo = -1
    for (const m of linkMarks) { lFrom = Math.min(lFrom, m.from); lTo = Math.max(lTo, m.to) }
    if (linkMarks.length && !nearCursor(state, lFrom, lTo))
      out.push({ from: lFrom, to: lTo, tag: "replace:LinkMarks", deco: Decoration.replace({}) })
    // URL 子节点折叠（光标在 URL 内时展开以便编辑）
    const c = node.node.cursor()
    if (c.firstChild()) do {
      if (c.name === "URL" && !nearCursor(state, c.from, c.to))
        out.push({ from: c.from, to: c.to, tag: "replace:URL", deco: Decoration.replace({}) })
    } while (c.nextSibling())
    out.push({ from: node.from, to: node.to, tag: "mark:omd-link", deco: Decoration.mark({ class: "omd-link" }) })
    return
  }
```

> 注：LinkMarks 整体折叠成一个 replace 区间的前提是 `[`、`]`、`(`、`)` 之间没有其他被折叠内容——若出现 RangeSet 重叠报错（Decoration.set 抛错），把 LinkMark 逐个折叠即可。测试会暴露。

```ts
import { EditorView, WidgetType } from "@codemirror/view"

export class CheckboxWidget extends WidgetType {
  constructor(readonly checked: boolean, readonly pos: number) { super() }
  eq(other: CheckboxWidget) { return this.checked === other.checked }
  toDOM() {
    const el = document.createElement("input")
    el.type = "checkbox"
    el.checked = this.checked
    el.className = "omd-checkbox"
    el.addEventListener("mousedown", e => e.preventDefault())
    el.addEventListener("click", e => {
      e.preventDefault()
      const view = EditorView.findFromDOM(el.parentElement!)
      if (!view) return
      const from = this.pos, to = from + 3   // "[ ]" / "[x]" 三个字符
      const insert = this.checked ? "[ ]" : "[x]"
      view.dispatch({ changes: { from, to, insert } })
    })
    return el
  }
  ignoreEvent() { return false }
}
```

```ts
import { nearCursor, type DecoSpec } from "./build"
import { CheckboxWidget } from "./widgets"

export function blockRules(node: SyntaxNodeRef, state: EditorState, out: DecoSpec[]) {
  switch (node.name) {
    case "TaskMark": {
      const text = state.doc.sliceString(node.from, node.to)
      const checked = /x/i.test(text)
      out.push({
        from: node.from, to: node.to, tag: "widget:checkbox",
        deco: Decoration.replace({ widget: new CheckboxWidget(checked, node.from) })
      })
      break
    }
    case "Blockquote": {
      // 给引用块每一行加样式
      for (let pos = node.from; pos <= node.to;) {
        const line = state.doc.lineAt(pos)
        out.push({ from: line.from, to: line.from, tag: "line:omd-blockquote", deco: Decoration.line({ class: "omd-blockquote" }) })
        pos = line.to + 1
      }
      break
    }
    case "BlockquoteMark": {
      const end = Math.min(node.to + 1, state.doc.lineAt(node.from).to)
      if (!nearCursor(state, node.from, end))
        out.push({ from: node.from, to: end, tag: "replace:BlockquoteMark", deco: Decoration.replace({}) })
      break
    }
    case "HorizontalRule":
      out.push({ from: node.from, to: node.from, tag: "line:omd-hr", deco: Decoration.line({ class: "omd-hr" }) })
      break
  }
}
```

> TaskMark 的 `to - from` 恒为 3（`[ ]`/`[x]`）；若 Lezer 版本把空格含进来导致宽度不同，用 `node.to - node.from` 代替硬编码 3 并调整 insert 逻辑——测试会暴露。

```ts
import { Compartment, StateEffect, StateField, TransactionSpec } from "@codemirror/state"
import { EditorView, keymap } from "@codemirror/view"
import { livePreviewPlugin } from "../decorations/build"

export function livePreviewExt() {
  return [livePreviewPlugin, EditorView.baseTheme({})]  // baseTheme 占位，样式由 desktop 主题提供
}

export const livePreviewCompartment = new Compartment()

export const isLivePreview = StateField.define<boolean>({
  create: () => true,
  update: (v, tr) => tr.effects.reduce((v, e) => e.is(toggleLivePreview) ? !v : v, v)
})

export const toggleLivePreview = StateEffect.define<undefined>()

// 把 toggle 翻译成 compartment 重配置
const switcher = EditorState.transactionExtender.of(tr => {
  if (!tr.effects.some(e => e.is(toggleLivePreview))) return null
  const now = !isLivePreview.get(tr.startState)
  return { effects: livePreviewCompartment.reconfigure(now ? livePreviewExt() : []) }
})
```

```tsx
import { useEffect, useRef, useState } from "react"
import { createEditor } from "./Editor"
import { invoke } from "@tauri-apps/api/core"
import { open, save } from "@tauri-apps/plugin-dialog"
import { keymap } from "@codemirror/view"
import { EditorView } from "@codemirror/view"
import "./styles.css"

export default function App() {
  const host = useRef<HTMLDivElement>(null)
  const viewRef = useRef<EditorView | null>(null)
  const [path, setPath] = useState<string | null>(null)
  const [dirty, setDirty] = useState(false)

  useEffect(() => {
    if (!host.current) return
    const view = createEditor(host.current)
    viewRef.current = view
    view.dom.addEventListener("input", () => setDirty(true))
    return () => { view.destroy(); viewRef.current = null }
  }, [])

  async function openFile() {
    const p = await open({ filters: [{ name: "Markdown", extensions: ["md", "markdown", "mdx"] }] })
    if (!p || typeof p !== "string") return
    const contents = await invoke<string>("read_file", { path: p })
    const view = viewRef.current!
    view.dispatch({ changes: { from: 0, to: view.state.doc.length, insert: contents } })
    setPath(p); setDirty(false)
  }

  async function saveFile() {
    if (!viewRef.current) return
    let p = path
    if (!p) {
      p = (await save({ filters: [{ name: "Markdown", extensions: ["md"] }] })) as string | null
      if (!p) return
    }
    await invoke("write_file", { path: p, contents: viewRef.current.state.doc.toString() })
    setPath(p); setDirty(false)
  }

  // Cmd+O / Cmd+S
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if (!e.metaKey) return
      if (e.key === "o") { e.preventDefault(); openFile() }
      if (e.key === "s") { e.preventDefault(); saveFile() }
    }
    window.addEventListener("keydown", handler)
    return () => window.removeEventListener("keydown", handler)
  })

  return (
    <div className="app">
      <div className="statusbar">{path ?? "untitled"}{dirty ? " •" : ""}</div>
      <div ref={host} className="editor-host" />
    </div>
  )
}
```
